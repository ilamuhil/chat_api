dashboard_db (public schema)

PostgreSQL types are shown using their database equivalents. Columns without
an explicit DEFAULT do not have a server-side default in the model. Python-side
enum values are noted separately; the model maps those columns as text and does
not create database CHECK constraints for them.


organizations
-------------
id                 text PRIMARY KEY
created_at         timestamptz NOT NULL DEFAULT now()
is_email_verified  boolean NOT NULL DEFAULT false
name               text
logo_url           text
address            jsonb
email              text
phone              text
email_token        text


bots
----
id                    uuid PRIMARY KEY DEFAULT gen_random_uuid()
created_at            timestamptz NOT NULL DEFAULT now()
name                  text NOT NULL
institute_name        text
capture_leads         boolean NOT NULL DEFAULT false
updated_at            timestamptz NOT NULL DEFAULT now()
organization_id       text REFERENCES organizations(id) ON DELETE CASCADE
tone                  text
role                  text
business_description  text
first_message         text
confirmation_message  text
lead_capture_message  text
lead_capture_timing   text
capture_name          boolean DEFAULT false
capture_email         boolean DEFAULT false
capture_phone         boolean DEFAULT false
deleted_at            timestamptz
deleted_by            text


users
-----
id                    uuid PRIMARY KEY DEFAULT gen_random_uuid()
created_at            timestamptz NOT NULL DEFAULT now()
updated_at            timestamptz NOT NULL DEFAULT now()
email                 text UNIQUE
email_verified        boolean NOT NULL DEFAULT false
email_verified_at     timestamptz
password_hash         text
phone                 varchar
phone_verified        boolean NOT NULL DEFAULT false
phone_verified_at     timestamptz
full_name             varchar
avatar_url             text
google_id             text UNIQUE
github_id             text UNIQUE
microsoft_id          text UNIQUE
google_email          varchar
github_email          varchar
microsoft_email       varchar
last_logged_in        timestamptz
last_logged_in_ip     varchar
is_active             boolean NOT NULL DEFAULT true
is_banned             boolean NOT NULL DEFAULT false
banned_until          timestamptz
onboarding_completed  boolean NOT NULL DEFAULT false

INDEX: users_email_idx
  ON (email)
INDEX: users_google_id_idx
  ON (google_id)
INDEX: users_github_id_idx
  ON (github_id)
INDEX: users_microsoft_id_idx
  ON (microsoft_id)


organization_members
--------------------
id                 uuid PRIMARY KEY DEFAULT gen_random_uuid()
created_at         timestamptz NOT NULL DEFAULT now()
organization_id    text REFERENCES organizations(id) ON DELETE CASCADE
role               text
user_id            uuid REFERENCES users(id) ON DELETE CASCADE

INDEX: organization_members_user_id_idx
  ON (user_id)
INDEX: organization_members_organization_id_idx
  ON (organization_id)


organization_invites
--------------------
id                 uuid PRIMARY KEY DEFAULT gen_random_uuid()
organization_id    text NOT NULL REFERENCES organizations(id) ON DELETE CASCADE
email              text NOT NULL
role               text NOT NULL DEFAULT 'editor'
created_at         timestamptz NOT NULL DEFAULT now()
accepted_at        timestamptz

INDEX: organization_invites_email_idx
  ON (email)
INDEX: organization_invites_organization_id_idx
  ON (organization_id)


otps
----
id                 uuid PRIMARY KEY DEFAULT gen_random_uuid()
created_at         timestamptz NOT NULL DEFAULT now()
user_id            uuid REFERENCES users(id) ON DELETE CASCADE
code               text NOT NULL
type               text NOT NULL
  -- Python enum values: EMAIL, MOBILE
purpose            text NOT NULL
  -- Python enum values: LOGIN, VERIFY_EMAIL, VERIFY_PHONE, RESET_PASSWORD
email              varchar
phone              varchar
expires_at         timestamptz NOT NULL
used_at            timestamptz
is_used            boolean NOT NULL DEFAULT false
attempts           integer NOT NULL DEFAULT 0
max_attempts       integer NOT NULL DEFAULT 5
ip_address         varchar
user_agent         text

INDEX: otps_user_id_idx
  ON (user_id)
INDEX: otps_email_idx
  ON (email)
INDEX: otps_phone_idx
  ON (phone)
INDEX: otps_code_idx
  ON (code)
INDEX: otps_expires_at_idx
  ON (expires_at)


api_keys
--------
id                 uuid PRIMARY KEY DEFAULT gen_random_uuid()
name               text NOT NULL
key_hash           text NOT NULL
created_at         timestamptz NOT NULL DEFAULT now()
organization_id    text REFERENCES organizations(id) ON DELETE CASCADE
bot_id             uuid REFERENCES bots(id) ON DELETE CASCADE

last_used_at       timestamptz DEFAULT now()
