python chat database schema (from app/models/chat_db_models.py)

PostgreSQL types are shown using their database equivalents. Columns without
an explicit DEFAULT do not have a server-side default in the model.


documents
---------
id                    uuid PRIMARY KEY DEFAULT gen_random_uuid()
organization_id       text NOT NULL
bot_id                uuid NOT NULL
source_id             uuid NOT NULL
chunk_index           integer NOT NULL
content               text
created_at            timestamptz DEFAULT now()
section_title         text
token_count           integer
is_active             boolean DEFAULT true
deleted_at            timestamptz
model_config_version_id uuid NOT NULL

UNIQUE (source_id, model_config_version_id, chunk_index)
  -- uq_documents_source_chunk_config_version

INDEX: documents_bot_active_model_config_version_idx
  ON (bot_id, is_active, model_config_version_id)
INDEX: documents_source_idx
  ON (source_id)


messages
--------
id                      uuid PRIMARY KEY
conversation_id         uuid NOT NULL
created_at              timestamptz DEFAULT now()
updated_at              timestamptz DEFAULT now()
role                    text
content_type            text NOT NULL DEFAULT 'text'
content                 varchar
model_config_version_id uuid

INDEX: messages_conversation_created_at_idx
  ON (conversation_id, created_at)


training_jobs
-------------
id                      uuid PRIMARY KEY
organization_id         text NOT NULL
bot_id                  uuid NOT NULL
status                  text NOT NULL
  -- queued, processing, completed, failed, cleanup_completed
started_at              timestamptz
completed_at            timestamptz
error_message           text
model_config_version_id uuid NOT NULL

INDEX: training_jobs_bot_status_idx
  ON (bot_id, status)
INDEX: training_jobs_model_config_version_idx
  ON (model_config_version_id)


embeddings (VECTOR STORE)
-------------------------
id          uuid PRIMARY KEY DEFAULT gen_random_uuid()
document_id uuid NOT NULL UNIQUE
embedding   vector(1536) NOT NULL
created_at  timestamptz NOT NULL DEFAULT now()
deleted_at  timestamptz

FOREIGN KEY (document_id) REFERENCES documents(id) ON DELETE CASCADE
  -- embeddings_document_id_fkey
INDEX: embeddings_document_id_idx
  ON (document_id)


retrieval_logs
--------------
id                      uuid PRIMARY KEY
organization_id         text
bot_id                  uuid
conversation_id         uuid
message_id              uuid
query                   text
query_embedding         vector(1536)
retrieved_document_ids  uuid[]
similarity_scores       double precision[]
retrieval_threshold     double precision
retrieval_k              integer
reranker_used            boolean
model_config_version_id uuid
reranked_document_ids   uuid[]
created_at              timestamptz DEFAULT now()

INDEX: retrieval_logs_bot_created_at_idx
  ON (bot_id, created_at)
INDEX: retrieval_logs_model_config_version_idx
  ON (model_config_version_id)


message_feedback
----------------
id         uuid PRIMARY KEY
message_id uuid
feedback   text CHECK (feedback IN ('positive', 'negative'))
reason     text
created_at timestamptz DEFAULT now()

FOREIGN KEY (message_id) REFERENCES messages(id)
  -- message_feedback_message_id_fkey


model_config_versions
---------------------
id                   uuid PRIMARY KEY
  -- Python-side default: uuid.uuid4
bot_id               uuid NOT NULL
version_number       integer NOT NULL
embedding_provider   text NOT NULL DEFAULT 'openai'
embedding_model      text NOT NULL
embedding_version    text
embedding_dimension  integer NOT NULL DEFAULT 1536
llm_provider         text NOT NULL
llm_model            text NOT NULL
reranker_provider    text
reranker_model       text
retrieval_k          integer NOT NULL
similarity_threshold double precision NOT NULL
chunk_size           integer NOT NULL
chunk_overlap       integer NOT NULL
active               boolean NOT NULL DEFAULT false
created_by_user_id   uuid
activated_at         timestamptz
created_at           timestamptz DEFAULT now()

UNIQUE (bot_id, version_number)
  -- model_config_version_bot_version_key

CHECK (retrieval_k > 0)
  -- model_config_versions_retrieval_k_positive
CHECK (similarity_threshold >= 0.0 AND similarity_threshold <= 1.0)
  -- model_config_versions_similarity_threshold_range
CHECK (chunk_size > 0)
  -- model_config_versions_chunk_size_positive
CHECK (chunk_overlap >= 0 AND chunk_overlap < chunk_size)
  -- model_config_versions_chunk_overlap_range
CHECK (embedding_dimension > 0)
  -- model_config_versions_embedding_dimension_positive

INDEX: model_config_versions_created_at_idx
  ON (bot_id, created_at)
UNIQUE INDEX: model_config_versions_one_active_per_bot_idx
  ON (bot_id) WHERE active = true
